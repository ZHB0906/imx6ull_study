## ---- 启动链关键项（已确认 ✓）----

# ★ initrd 启动链（当前生效 ✓）：先 fatload 三个文件 → 设 bootargs → bootz → 失败则 run emmcboot
#   ★ 注意 bootargs 里的 ip= 已被去掉（改为用户态 S39eth0 配 eth0 ✓ 见文档 §20 ✓）
bootcmd=fatload mmc 1:1 83800000 initrd; fatload mmc 1:1 80800000 zImage; fatload mmc 1:1 83000000 imx6ull-14x14-emmc-7-1024x600-c.dtb; setenv bootargs console=ttymxc0,115200 root=/dev/mmcblk1p2 rootwait rw panic=10; bootz 80800000 83800000:0x1A2C82 83000000; run emmcboot

# ★ 兜底（kmcboot）：从 eMMC 直接起（不带 initrd）✓ 一定保留 ✓
#   （原文见 out/uboot-bootcmd-before-initrd.txt ✓）
bootargs=console=ttymxc0,115200 root=/dev/mmcblk1p2 rootwait rw
ipaddr=192.168.10.2
serverip=192.168.10.1
ethaddr=00:04:9f:04:d2:36
netretry=no
bootdelay=5
